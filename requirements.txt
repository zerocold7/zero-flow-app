# What the engine needs. Install it together with constraints.txt, which pins every
# package underneath these to a set known to work together:
#   pip install -r requirements.txt -c constraints.txt
# Without it pip takes the newest of each, and a newer transformers (5.19) broke the
# reader on a fresh install while this file had not changed at all.

# --- NVIDIA GPU acceleration (CUDA 12) ---
# Required for faster-whisper device="cuda". Without these, CTranslate2 cannot
# load cublas64_12.dll and transcription fails. PCs without an NVIDIA card skip
# them (docs/INSTALL.md step 4). cuDNN 9.x requires ctranslate2 >= 4.5.
nvidia-cublas-cu12~=12.4
nvidia-cudnn-cu12~=9.1

# --- Reader: text-to-speech (Kokoro) ---
# kokoro pulls in torch. On an NVIDIA PC install the CUDA build of torch FIRST
# (docs/INSTALL.md step 4), or pip takes the CPU-only one. The reader runs on the
# CPU too — Kokoro-82M is small.
kokoro==0.9.4
# Kokoro's English pronunciation needs spaCy's small English model. Without it here,
# it is pip-installed at runtime on the first read — a hidden download, and a failure
# when offline.
en_core_web_sm @ https://github.com/explosion/spacy-models/releases/download/en_core_web_sm-3.8.0/en_core_web_sm-3.8.0-py3-none-any.whl#sha256=1932429db727d4bff3deed6b34cfc05df17794f4a52eeb26cf8928f7c1a0fb85
